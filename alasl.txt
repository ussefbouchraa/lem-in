var blocked = make(map[string]bool)

func BfsWithBlock(connections map[string][]string, start, end string, blocked map[string]bool) []string {
	queue := [][]string{{start}}
	visited := map[string]bool{start: true}

	for len(queue) > 0 {
		// fmt.Println("que--> ", queue)
		path := queue[0]
		queue = queue[1:]
		last := path[len(path)-1]

		if last == end {
			return path
		}

		for _, neighbor := range connections[last] {

			if visited[neighbor] || blocked[neighbor] {
				continue
			}
			visited[neighbor] = true
			newPath := append([]string{}, path...)
			newPath = append(newPath, neighbor)
			queue = append(queue, newPath)
		}
	}
	return nil
}

func FindAllDisjointPaths(connections map[string][]string, start, end string) [][]string {
	var allPaths [][]string

	for {
		path := BfsWithBlock(connections, start, end, blocked)
		if path == nil {
			break
		}
		allPaths = append(allPaths, path)

		for _, node := range path {
			if node != start && node != end {
				blocked[node] = true
			}
		}
	}

	return allPaths
}
